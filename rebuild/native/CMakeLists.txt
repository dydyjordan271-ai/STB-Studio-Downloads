cmake_minimum_required(VERSION 3.24)
project(STBMesh LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(RAGEKIT_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(RAGEKIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RAGEKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RAGEKIT_BUILD_RENDER OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(ragekit GIT_REPOSITORY https://github.com/RatPrez/ragekit.git GIT_TAG v0.3.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(ragekit)
add_executable(stb-mesh stb-mesh.cpp)
target_link_libraries(stb-mesh PRIVATE ragekit::ragekit)
set_target_properties(stb-mesh PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/STB")
